## Ejercicio 4.1 · Expresión if

```kotlin
val mayor = if (a > b) a else b
```

## Ejercicio 4.2 · Clasificación con when

```kotlin
fun clasificar(nota: Int): String = when {
    nota >= 9 -> "Sobresaliente"
    nota >= 7 -> "Notable"
    nota >= 5 -> "Aprobado"
    else -> "Suspenso"
}
```

## Ejercicio 4.3 · Bucle while

```kotlin
var contador = 1
while (contador <= 5) {
    println(contador)
    contador++
}
```

## Ejercicio 4.4 · Bucle do-while

```kotlin
var entrada: String
do {
    print("Escribe salir: ")
    entrada = readln()
} while (entrada != "salir")
```

## Ejercicio 4.5 · Recorrer rango

```kotlin
for (numero in 1..10) println(numero)
```

## Ejercicio 4.6 · Saltos de iteración

```kotlin
for (numero in 1..10) {
    if (numero % 2 == 0) continue
    if (numero > 7) break
    println(numero)
}
```

## Ejercicio 4.7 · Índices y valores

```kotlin
val nombres = arrayOf("Ana", "Luis", "Eva")
for ((indice, nombre) in nombres.withIndex()) {
    println("$indice: $nombre")
}
```

## Ejercicio 4.8 · Rango descendente

```kotlin
for (numero in 10 downTo 0 step 2) println(numero)
```

## Ejercicio 4.9 · Media de valores

```kotlin
val notas = doubleArrayOf(8.0, 6.5, 9.0)
var suma = 0.0
for (nota in notas) suma += nota
println(suma / notas.size)
```

## Ejercicio 4.10 · Tabla de multiplicar

```kotlin
val numero = 7
for (multiplicador in 1..10) {
    println("$numero × $multiplicador = ${numero * multiplicador}")
}
```
